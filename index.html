// פונקציית טעינה מעודכנת שתומכת בעברית async function fetchTasks() { const token = document.getElementById('ghToken').value || localStorage.getItem('gh_token'); if (!token) return; try { const res = await fetch(`https://api.github.com/repos/${REPO}/contents/${FILE_PATH}`, { headers: { 'Authorization': `token ${token}`, 'Cache-Control': 'no-cache' } }); const data = await res.json(); sha = data.sha; // פענוח בטוח של עברית מ-Base64 const binaryString = atob(data.content); const bytes = new Uint8Array(binaryString.length); for (let i = 0; i < binaryString.length; i++) { bytes[i] = binaryString.charCodeAt(i); } const decodedText = new TextDecoder('utf-8').decode(bytes); const tasks = JSON.parse(decodedText); renderTasks(tasks); } catch (e) { console.error("Error loading tasks", e); } } // פונקציית שמירה מעודכנת async function updateGithub(newTasks, token, currentSha) { document.getElementById('status').innerText = "שומר בגיטהאב..."; try { const jsonString = JSON.stringify(newTasks, null, 2); // קידוד UTF-8 בטוח לעברית const bytes = new TextEncoder().encode(jsonString); const base64Content = btoa(String.fromCharCode(...bytes)); const res = await fetch(`https://api.github.com/repos/${REPO}/contents/${FILE_PATH}`, { method: 'PUT', headers: { 'Authorization': `token ${token}`, 'Content-Type': 'application/json' }, body: JSON.stringify({ message: "Update tasks via App", content: base64Content, sha: currentSha }) }); if (res.ok) { document.getElementById('status').innerText = "נשמר בהצלחה!"; document.getElementById('taskInput').value = ''; fetchTasks(); } else { const err = await res.json(); document.getElementById('status').innerText = "שגיאה: " + err.message; } } catch (e) { document.getElementById('status').innerText = "שגיאה בחיבור"; } }
